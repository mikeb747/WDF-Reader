# ------------------------------------------------------------
# 1. Locate system folder inside My Documents
# ------------------------------------------------------------

$docs = [Environment]::GetFolderPath("MyDocuments")

# Find the first folder that looks like a system folder (e.g., C14R96_FZJ)
$systemFolder = Get-ChildItem -Path $docs -Directory |
    Where-Object { $_.Name -match "^[0-9A-Z]+_" } |
    Select-Object -First 1

if (-not $systemFolder) {
    Write-Host "No system folder found in My Documents." -ForegroundColor Red
    return
}

$folderPrefix = $systemFolder.Name.Substring(0,6)

Write-Host "System folder detected: $($systemFolder.FullName)" -ForegroundColor Cyan

# Find newest *_sys.rrg file (search all subfolders)
$rrgFile = Get-ChildItem -Path $systemFolder.FullName -Recurse -Filter "*_sys.rrg" |
    Sort-Object LastWriteTime -Descending |
    Select-Object -First 1

if (-not $rrgFile) {
    Write-Host "No *_sys.rrg file found in system folder." -ForegroundColor Red
    return
}

Write-Host "RRg file detected: $($rrgFile.FullName)" -ForegroundColor Cyan

# ------------------------------------------------------------
# 2. Load file
# ------------------------------------------------------------

$lines = Get-Content $rrgFile.FullName | ForEach-Object { $_.ToString() }

# ------------------------------------------------------------
# 3. Generic block reader (no regex for [HKEY_])
# ------------------------------------------------------------

function Get-RRGBlockValue {
    param(
        [string[]]$lines,
        [string]$path   # e.g. "Laser Collection\laser0\LaserName"
    )

    $header = "[HKEY_LOCAL_MACHINE\SOFTWARE\Renishaw\WiRE2\Systems\System0\$path]"

    for ($i = 0; $i -lt $lines.Count; $i++) {

        if ($lines[$i] -eq $header) {

            for ($j = $i + 1; $j -lt $lines.Count; $j++) {

                if ($lines[$j].StartsWith("[")) { break }

                if ($lines[$j] -match '"__value__"="([^"]+)"') {
                    return $Matches[1]
                }
            }
        }
    }

    return ""
}

# ------------------------------------------------------------
# 4. Instrument Serial Number + Test 5
# ------------------------------------------------------------

$instrumentSerial = Get-RRGBlockValue $lines "GeneralProperties\Instrument Serial Number"
Write-Host "Instrument Serial Number: $instrumentSerial" -ForegroundColor Green

$serialPrefix = if ($instrumentSerial.Length -ge 6) { $instrumentSerial.Substring(0,6) } else { "" }

$test5_pass = $true
if ($folderPrefix -ne $serialPrefix) {
    Write-Host "Error: Folder prefix ($folderPrefix) does not match Instrument Serial Number ($serialPrefix)" -ForegroundColor Red
    $test5_pass = $false
}

# ------------------------------------------------------------
# 5. Find all laser folders (laser0, laser1, etc.)
# ------------------------------------------------------------

$laserIndexes = for ($i = 0; $i -lt $lines.Count; $i++) {
    if ($lines[$i] -match "Laser Collection\\laser([0-9]+)\\LaserName]\s*$") {
        [PSCustomObject]@{
            Index       = $i
            LaserFolder = "laser$($Matches[1])"
        }
    }
}

# ------------------------------------------------------------
# 6. Build results table
# ------------------------------------------------------------

$results = @()

foreach ($laser in $laserIndexes) {

    $folder = $laser.LaserFolder

    $laserName        = Get-RRGBlockValue $lines "Laser Collection\$folder\LaserName"
    $wavenumberString = Get-RRGBlockValue $lines "Laser Collection\$folder\LaserWavenumber"
    $maxPowerString   = Get-RRGBlockValue $lines "Laser Collection\$folder\MaxPower"
    $ndPercent        = Get-RRGBlockValue $lines "Laser Collection\$folder\NDPercent"

    $wavenumber = 0
    [double]::TryParse($wavenumberString, [ref]$wavenumber) | Out-Null

    $maxPower = 0
    [double]::TryParse($maxPowerString, [ref]$maxPower) | Out-Null

    # Example: fixed grating path for Static Dynamic Offset
    $staticDynamicOffset = Get-RRGBlockValue $lines "Grating Collection\grating0\1\0\Static Dynamic Offset"

    $results += [PSCustomObject]@{
        LaserFolder         = $folder
        LaserName           = $laserName
        Wavenumber          = $wavenumber
        MaxPower            = $maxPower
        NDPercent           = $ndPercent
        StaticDynamicOffset = $staticDynamicOffset
    }
}

# ------------------------------------------------------------
# 7. Output table
# ------------------------------------------------------------

Write-Host "`n=== VIRSA LASER TABLE ===" -ForegroundColor Cyan
Write-Host ("{0,-10} {1,-15} {2,-12} {3,-12} {4,-8} {5,-15}" -f "Key", "Laser Name", "Wavelength", "Wavenumber", "Power", "DynamicOffset") -ForegroundColor Yellow
Write-Host ("{0,-10} {1,-15} {2,-12} {3,-12} {4,-8} {5,-15}" -f "------", "----------", "----------", "----------", "-----", "--------------") -ForegroundColor Yellow

# Initialise test flags
$test1_pass = $true   # Default wavenumber check
$test2_pass = $true   # HiRes/HiSens match check
$test3_pass = $true   # Wavelength + tolerance
$test4_pass = $true   # Power > 12

# Add logic columns (extract wavelength)
$results | ForEach-Object {
    if ($_.LaserName -match "(\d{3})") {
        $_ | Add-Member -NotePropertyName Wavelength -NotePropertyValue $Matches[1]
    }
    else {
        $_ | Add-Member -NotePropertyName Wavelength -NotePropertyValue ""
        $test3_pass = $false
    }
} | Out-Null

# --- Default wavenumber lookup table (by wavelength) ---
$defaultWNs = @{
    "405" = 24691.4
    "532" = 18797
    "660" = 15151.5
    "785" = 12738.9
    "830" = 12048.2
}

# --- Tolerance table (by wavelength) ---
$targets = @{
    "532" = @{ Target = 18793.46; Tol = 10 }
    "785" = @{ Target = 12742.1;  Tol = 2  }
}

foreach ($row in $results) {

    $wl = $row.Wavelength

    # --- Default wavenumber test ---
    $isDefault = $false
    if ($defaultWNs.ContainsKey($wl)) {
        $defaultWN = $defaultWNs[$wl]
        $isDefault = ($row.Wavenumber -eq $defaultWN)
        if ($isDefault) { $test1_pass = $false }
    }
    else {
        Write-Host "Warning: New Laser Detected — no default wavenumber set for $wl nm" -ForegroundColor Yellow
    }

    # --- Tolerance test ---
    $wnFail = $false
    if ($targets.ContainsKey($wl)) {
        $target = $targets[$wl].Target
        $tol    = $targets[$wl].Tol

        if ([math]::Abs($row.Wavenumber - $target) -gt $tol) {
            $wnFail = $true
            $test3_pass = $false
        }
    }
    else {
        Write-Host "Warning: New Laser Detected — no tolerance range set for $wl nm" -ForegroundColor Yellow
    }

    # --- Power test ---
    $lowPower = $row.MaxPower -le 12
    if ($lowPower) { $test4_pass = $false }

    # --- Print row ---
    if ($isDefault -or $wnFail -or $lowPower) {

        Write-Host ("{0,-10} {1,-15} {2,-12} " -f $row.LaserFolder, $row.LaserName, $wl) -NoNewline

        if ($isDefault -or $wnFail) {
            Write-Host ("{0,-12}" -f $row.Wavenumber) -ForegroundColor Red -NoNewline
        }
        else {
            Write-Host ("{0,-12}" -f $row.Wavenumber) -NoNewline
        }

        if ($lowPower) {
            Write-Host ("{0,-8}" -f $row.MaxPower) -ForegroundColor Red
        }
        else {
            Write-Host ("{0,-8}" -f $row.MaxPower)
        }

        if ($isDefault) { Write-Host "Error: Default Laser Wave Number" -ForegroundColor Red }
        if ($wnFail)    { Write-Host "Error: Wavenumber out of tolerance for $wl nm" -ForegroundColor Red }
        if ($lowPower)  { Write-Host "Error: Laser power too low (must be > 12)" -ForegroundColor Red }

    }
    else {
        "{0,-10} {1,-15} {2,-12} {3,-12} {4,-8} {5,-15}" -f `
            $row.LaserFolder, $row.LaserName, $wl, $row.Wavenumber, $row.MaxPower, $row.StaticDynamicOffset
    }
}

# ------------------------------------------------------------
# 8. Check HiRes / HiSens pairs match (Test 2)
# ------------------------------------------------------------

$grouped = $results | Group-Object Wavelength

foreach ($g in $grouped) {

    if ($g.Count -lt 2) {
        Write-Host "Error: Missing Laser Port HiRes/HiSens for $($g.Name) nm" -ForegroundColor Red
        $test2_pass = $false
        continue
    }

    $wn1 = $g.Group[0].Wavenumber
    $wn2 = $g.Group[1].Wavenumber

    if ($wn1 -ne $wn2) {
        Write-Host "Error: HiRes / HiSens mismatch for $($g.Name) nm" -ForegroundColor Red
        $test2_pass = $false
    }
}

# ------------------------------------------------------------
# 9. Print test summary footer
# ------------------------------------------------------------

Write-Host "`n=== TEST SUMMARY ===" -ForegroundColor Cyan

if ($test1_pass) { Write-Host "Test 1 Passed (No default wavenumbers detected)" -ForegroundColor Green }
else             { Write-Host "Test 1 FAILED (Default wavenumber found)" -ForegroundColor Red }

if ($test2_pass) { Write-Host "Test 2 Passed (HiRes&HiSens wavenumbers match)" -ForegroundColor Green }
else             { Write-Host "Test 2 FAILED (HiRes&HiSens wavenumbers mismatch)" -ForegroundColor Red }

if ($test3_pass) { Write-Host "Test 3 Passed (Wavelength + tolerance OK)" -ForegroundColor Green }
else             { Write-Host "Test 3 FAILED (Wavelength or tolerance issue)" -ForegroundColor Red }

if ($test4_pass) { Write-Host "Test 4 Passed (All lasers have power > 12)" -ForegroundColor Green }
else             { Write-Host "Test 4 FAILED (One or more lasers have low power)" -ForegroundColor Red }

if ($test5_pass) { Write-Host "Test 5 Passed (Folder name matches Instrument Serial Number)" -ForegroundColor Green }
else             { Write-Host "Test 5 FAILED (Folder name does NOT match Instrument Serial Number)" -ForegroundColor Red }
