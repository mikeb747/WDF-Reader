# region: Helpers and reference maps
$script:PropertyMap = if (Test-Path "$PSScriptRoot\WdfPropertyKeys.txt") { Get-Content -Path "$PSScriptRoot\WdfPropertyKeys.txt" -Raw | ConvertFrom-StringData } else { @{} }

$script:BlockMap = @{}
if (Test-Path "$PSScriptRoot\BlockMap.txt") {
    Get-Content -Path "$PSScriptRoot\BlockMap.txt" | ForEach-Object {
        if ($_ -match "(0x[0-9A-Fa-f]+)\s*=\s*`"(.+?)`"") {
            $HexID = [long][System.Convert]::ToUInt32($matches[1], 16)
            $script:BlockMap[$HexID] = $matches[2]
        }
    }
}

function Get-WdfBlockName {
    param([uint32]$Id)
    if ($script:BlockMap.ContainsKey([long]$Id)) {
        return $script:BlockMap[[long]$Id]
    }
    "Unknown (0x{0:X8})" -f $Id
}
# region: Core PSET parser

function Convert-WdfPset {
    param(
        [byte[]]$Bytes,
        [int]$StartOffset
    )

    $marker = [System.BitConverter]::ToUInt32($Bytes, $StartOffset)
    if ($marker -ne 0x54455350) { return $null } # 'PSET'

    $psetSize  = [System.BitConverter]::ToUInt32($Bytes, $StartOffset + 4)
    $psetStart = $StartOffset + 8
    $psetEnd   = $psetStart + $psetSize

    $entries = [ordered]@{}
    $offset  = $psetStart

    while ($offset -lt $psetEnd - 8) {
        $key    = [System.BitConverter]::ToUInt16($Bytes, $offset)
        $type   = [System.BitConverter]::ToUInt16($Bytes, $offset + 2)
        $length = [System.BitConverter]::ToUInt32($Bytes, $offset + 4)
        $valueOffset = $offset + 8

        if ($valueOffset + $length -gt $psetEnd) { break }

        $hexKey = ('0x{0:X4}' -f $key)
        $name   = if ($script:PropertyMap.ContainsKey($hexKey)) { $script:PropertyMap[$hexKey] } else { $hexKey }

        $value = $null
        switch ($length) {
            4 {
                $value = [PSCustomObject]@{
                    Int32  = [System.BitConverter]::ToInt32($Bytes, $valueOffset)
                    Float  = [System.BitConverter]::ToSingle($Bytes, $valueOffset)
                    Type   = $type
                }
            }
            8 {
                $value = [PSCustomObject]@{
                    Int64  = [System.BitConverter]::ToInt64($Bytes, $valueOffset)
                    Double = [System.BitConverter]::ToDouble($Bytes, $valueOffset)
                    Type   = $type
                }
            }
            default {
                $str = [System.Text.Encoding]::UTF8.GetString($Bytes, $valueOffset, $length).TrimEnd([char]0)
                $value = [PSCustomObject]@{
                    String = $str
                    Type   = $type
                }
            }
        }

        if ($entries.Contains($name)) {
            if (-not ($entries[$name] -is [System.Collections.IList])) {
                $entries[$name] = [System.Collections.Generic.List[object]]@($entries[$name])
            }
            $entries[$name].Add($value)
        } else {
            $entries[$name] = $value
        }

        $offset = $valueOffset + [int]$length
    }

    [PSCustomObject]@{
        PsetSize  = $psetSize
        Entries   = $entries
        EndOffset = $psetEnd
    }
}

# region: Block readers

function Read-WdfHeader {
    param([byte[]]$Bytes, [int]$Offset)

    # Offset 60: number of points per spectrum (uint32) [Source 306]
    $npoints = [System.BitConverter]::ToUInt32($Bytes, $Offset + 60)
    
    $title = [System.Text.Encoding]::UTF8.GetString($Bytes, $Offset + 240, 160).TrimEnd("`0")
    $user  = [System.Text.Encoding]::UTF8.GetString($Bytes, $Offset + 208, 32).TrimEnd("`0")
    $laser = [System.BitConverter]::ToSingle($Bytes, $Offset + 156)

    [PSCustomObject]@{
        npoints    = $npoints
        Title      = $title
        Operator   = $user
        Laser      = $laser
    }
}

function Read-WdfDataBlock {
    param(
        [byte[]]$Bytes,
        [int]$Offset,
        [uint64]$Size
    )

    $dataStart = $Offset + 16
    $dataBytes = [int]$Size - 16
    if ($dataBytes -lt 0) { return @() }

    $count = [int]($dataBytes / 4)
    $values = New-Object float[] $count
    for ($i = 0; $i -lt $count; $i++) {
        $values[$i] = [System.BitConverter]::ToSingle($Bytes, $dataStart + ($i * 4))
    }
    $values
}

function Read-WdfAxisBlock {
    param(
        [byte[]]$Bytes,
        [int]$Offset,
        [uint64]$Size
    )

    $dataStart = $Offset + 16
    $dataBytes = [int]$Size - 16
    if ($dataBytes -lt 0) { return @() }

    $count = [int]($dataBytes / 8)
    $values = New-Object double[] $count
    for ($i = 0; $i -lt $count; $i++) {
        $values[$i] = [System.BitConverter]::ToDouble($Bytes, $dataStart + ($i * 8))
    }
    $values
}

function Read-WdfPsetsInBlock {
    param(
        [byte[]]$Bytes,
        [int]$Offset,
        [uint64]$Size
    )

    $blockEnd = $Offset + [int]$Size
    if ($blockEnd -gt $Bytes.Length) { $blockEnd = $Bytes.Length }

    $psets = @()
    $i = $Offset + 16
    while ($i -lt ($blockEnd - 8)) {
        $marker = [System.BitConverter]::ToUInt32($Bytes, $i)
        if ($marker -eq 0x54455350) {
            $pset = Convert-WdfPset -Bytes $Bytes -StartOffset $i
            if ($pset) {
                $psets += $pset
                $i = $pset.EndOffset
                continue
            }
        }
        $i++
    }
    $psets
}

# region: High-level parser

function Get-WdfSpectrum {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory, ValueFromPipeline, ValueFromPipelineByPropertyName)]
        [string]$Path
    )

    process {
        if (-not (Test-Path $Path)) { throw "File not found: $Path" }


        $bytes = [System.IO.File]::ReadAllBytes((Resolve-Path $Path))
        $offset = 0

        $header, $xAxis, $yAxis      = $null, $null, $null
        $instrument = @{}
        $Metadata   = [ordered]@{} # Renamed and initialized correctly

        while ($offset -lt $bytes.Length) {
            $blockId   = [System.BitConverter]::ToUInt32($bytes, $offset)
            #$blockUid  = [System.BitConverter]::ToUInt32($bytes, $offset + 4)
            $blockSize = [System.BitConverter]::ToUInt64($bytes, $offset + 8)

            if ($blockSize -le 0) { break }

            switch ($blockId) {
                0x31464457 { # WdfHeader
                    $header = Read-WdfHeader -Bytes $bytes -Offset $offset
                }
                
                0x41544144 { # DATA (Spectrum Intensities)
                    $SpectrumData = Read-WdfDataBlock -Bytes $bytes -Offset $offset -Size $blockSize
                }

            0x54534C58 { # XLST (X-Axis List)
                if ($header.npoints -gt 0) {
                    $axis = New-Object float[] $header.npoints
                    # Change from 28 to 24 to pick up 1392.67
                    $dataStart = $offset + 24 
                    for ($i = 0; $i -lt $header.npoints; $i++) {
                        $axis[$i] = [System.BitConverter]::ToSingle($bytes, $dataStart + ($i * 4))
                    }
                    $xAxis = $axis
                }
            }

                0x54534C59 { # YLST (Y-Axis List)
                    if ($header.npoints -gt 0) {
                        $yAxis = New-Object float[] $header.npoints
                        # Offset 40: skips 16-byte block header + 24 bytes (6 floats) of junk metadata
                        $dataStart = $offset + 40 
                        for ($i = 0; $i -lt $header.npoints; $i++) {
                            $yAxis[$i] = [System.BitConverter]::ToSingle($bytes, $dataStart + ($i * 4))
                        }
                        $yAxisCoords = $yAxis
                    }
                }

                0x53495857 { # WXIS Instrument Settings
                    $psets = Read-WdfPsetsInBlock -Bytes $bytes -Offset $offset -Size $blockSize
                    $instr = @{}

                    foreach ($pset in $psets) {
                        foreach ($entry in $pset.Entries.Values) {

                            $values = if ($entry -is [System.Collections.IEnumerable] -and -not ($entry -is [string])) {
                                $entry
                            } else {
                                @($entry)
                            }

                            foreach ($v in $values) {
                                if ($v.PSObject.Properties.Name -contains 'String') {

                                    $raw = $v.String
                                    if (-not $raw) { continue }

                                    # Extract ASCII substrings (2+ chars)
                                    $ascii = [regex]::Matches($raw, '[\x20-\x7E]{2,}') |
                                            ForEach-Object { $_.Value }

                                    foreach ($token in $ascii) {

                                        # Clean trailing Renishaw markers
                                        $tokenClean = $token -replace '[ukiq]+$', ''
                                        $tokenClean = $tokenClean.Trim()

                                        # Grating
                                        if ($tokenClean -match '\d{3,4}\s*l\/mm') {
                                            $instr.Grating = $tokenClean
                                        }

                                        # Laser line
                                        elseif ($tokenClean -match '\d{3,4}\s*nm') {
                                            $instr.LaserLine = $tokenClean
                                        }

                                        # Objective
                                        elseif ($tokenClean -match '^\d{1,3}x$') {
                                            $instr.Objective = $tokenClean
                                        }

                                        # Laser index (laser0, laser1…)
                                        elseif ($tokenClean -match '^laser\d+$') {
                                            $instr.LaserIndex = $tokenClean
                                        }

                                        # Laser power (65mW etc.)
                                        elseif ($tokenClean -match '^\d+(\.\d+)?mW$') {
                                            $instr.LaserPower = $tokenClean
                                        }

                                        # Grating index (grating0, grating1…)
                                        elseif ($tokenClean -match '^grating\d+$') {
                                            $instr.GratingIndex = $tokenClean
                                        }
                                    }
                                }
                            }
                        }
                    }

                    $instrument = $instr
                }

                0x4D445857 { # WXDM Measurement Settings (Scan configuration)
                    $psets = Read-WdfPsetsInBlock -Bytes $bytes -Offset $offset -Size $blockSize
                    foreach ($pset in $psets) {
                        foreach ($entry in $pset.Entries.GetEnumerator()) {
                            $keyName = $entry.Key # Already formatted as "ExposureTime" or "0x0069" by Convert-WdfPset
                            $v = $entry.Value

                            # 1. Skip massive VBScript data streams [Source 280]
                            if ($v.Type -eq 1004) { continue }

                            # 2. Handle Strings (Extract and Clean)
                            if ($v.PSObject.Properties.Name -contains 'String') {
                                # Clean trailing Renishaw markers and nulls [Source 277]
                                $strClean = $v.String -replace '[ukiq]+$', ''
                                $strClean = $strClean.Trim().TrimEnd("`0")
                                
                                if (-not $strClean) { continue }
                                $Metadata[$keyName] = $strClean

                                # Extract numeric fields if they are embedded in strings
                                if ($strClean -match '(\d+)\s*sec') { $Metadata['ExposureTime'] = [int]$matches[4] }
                                if ($strClean -match '(\d+)\s*accum') { $Metadata['Accumulations'] = [int]$matches[4] }
                            }
                            # 3. Handle Numeric Values (Unpack from PSET Object) [Source 270]
                            else {
                                $val = if ($v.Float -ne $null) { $v.Float } else { $v.Int32 }
                                if ($val -ne $null) { $Metadata[$keyName] = $val }
                            }
                        }
                    }
                }




                default {
                    # Other blocks ignored for now (ORGN, WXDA, WXCS, ZLDC, TEXT, etc.)
                }
            }

            $offset += [int]$blockSize
        }

# Update the return block at the end of Get-WdfSpectrum
return [PSCustomObject]@{
    Path       = $Path
    Title      = $header.Title      # 0x31464457 WdfHeader
    Operator   = $header.Operator   # 0x31464457 WdfHeader
    PointCount = $header.npoints    # 0x31464457 WdfHeader
    Laser      = $header.Laser      # 0x31464457 WdfHeader
    Instrument = $instrument        # 0x53495857 WXIS Instrument Settings (Hardware state)
    Metadata   = $Metadata          # 0x4D445857 WXDM Measurement Settings (Scan configuration)
    XAxis      = $xAxis             # 0x54534C58 XLST (X-Axis List)
    YAxis      = $yAxisCoords       # 0x54534C59 YLST (Y-Axis List)
    Spectrum   = $SpectrumData      # 0x41544144 DATA (Spectral Data)
}
} 
}
